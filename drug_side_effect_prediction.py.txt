# Load the dataset
data = pd.read_csv('drugs_side_effects.csv')

# Split the data into drug names and side effects
drug_names = data['drug_name'].values
side_effects=data['side_effects'].values
nltk.download('punkt')
nltk.download('stopwords')
k=[]
for i in range(0,len(side_effects)):
  k.append(word_tokenize(side_effects[i]))
from nltk.corpus import stopwords
l=[]
for i in range(0,len(k)):
  t=[]
  for word in k[i]:
    if word not in stopwords.words('english') and len(word)!=1:
      t.append(word)
  l.append(t)
# Tokenize the drug names and side effects
tokenizer_drug = Tokenizer()
tokenizer_drug.fit_on_texts(drug_names)

tokenizer_side = Tokenizer()
tokenizer_side.fit_on_texts(l)
#print(side_effects)
# Convert the drug names and side effects to sequences
drug_sequences = tokenizer_drug.texts_to_sequences(drug_names)
side_effects_sequences = tokenizer_side.texts_to_sequences(l)
print(side_effects_sequences)
# Pad the sequences to a maximum length
maxLen=0
for i in range(0,len(l)):
  if len(l[i])>maxLen:
    maxLen=len(l[i])
drug_data = pad_sequences(drug_sequences, maxlen=1)
print(drug_data)
side_effects_data = pad_sequences(side_effects_sequences, maxlen=maxLen)
print(side_effects_data)
# Convert the side effects sequences to one-hot encoding
num_classes = len(tokenizer_side.word_index) + 1
side_effects_labels = np.zeros((len(data), num_classes))
for i, label in enumerate(side_effects_sequences):
    side_effects_labels[i, label] = 1
print(side_effects_labels)


X_train_val, X_test, y_train_val, y_test = train_test_split(drug_data, side_effects_labels, test_size=0.2, random_state=42)

# Build the RNN model
model = Sequential()
model.add(Embedding(input_dim=len(tokenizer_drug.word_index) + 1, output_dim=64, input_length=100))
model.add(LSTM(64))
model.add(Dense(num_classes, activation='sigmoid'))

# Compile the model
model.compile(loss='binary_crossentropy', optimizer='adam', metrics=['accuracy'])

# Train the model
history=model.fit(X_train, y_train, batch_size=32, epochs=4, validation_data=(X_val, y_val))

def predict_side_effects(drug_name, model, tokenizer_drug, maxlen, num_classes):
    # Tokenize the drug name
    drug_sequence = tokenizer_drug.texts_to_sequences([drug_name])
    drug_data = pad_sequences(drug_sequence, maxlen=maxlen)

    # Predict the side effects
    predictions = model.predict(drug_data)

    # Convert the predictions to labels
    predicted_labels = []
    for i in range(len(predictions[0])):
        if predictions[0][i] > 0.4:
            predicted_labels.append(tokenizer_side.index_word[i])

    return predicted_labels
